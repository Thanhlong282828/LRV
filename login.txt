#[Hidden(['password', 'remember_token'])]
class User extends Authenticatable
{
    /** @use HasFactory<UserFactory> */
    use HasFactory, Notifiable;

    /**
     * Get the attributes that should be cast.
     * @ var list<string>
     * cho phép dc tương tác các cột trong DB
     * @return array<string, string>
     */
    protected $fillable = [
        'name',
        'email',
        'password',
        'birthday',
        'sex',
        'part_id',
        'position_id',
        'type_work',
        'team_id',
        'phone',
        'address',
        'status',
        'start_day',
        'end_day',
        'type_account_id',

    ];

    protected $hidden = [
        'password',
        'remember_token',
    ];

// thêm function để thể hiện các mối quan hệ
// tạo váo Models/Part để ánh xạ với bảng parts trongDB
public function part(){ 
    return $this->belongsTo(Part::class, 'part_id');
}
// tạo váo Models/Position để ánh xạ với bảng positions trongDB
    public function position(){
    return $this->belongsTo(Position::class, 'position_id');
}
// tạo váo Models/Team để ánh xạ với bảng teams trongDB
    public function team(){
    return $this->belongsTo(Team::class, 'team_id');
}
// tạo váo Models/typeAccount để ánh xạ với bảng type_accounts trongDB
    public function typeAccount(){
        return $this->belongsTo(TypeAccount::class, 'type_account_id');
}

    protected function casts(): array
    {
        return [
            'email_verified_at' => 'datetime',
            'password' => 'hashed',
        ];
    }
}

 1. whereHas() — lọc theo bảng quan hệ
 Ví dụ: lấy những User có ít nhất một Order trạng thái paid.
 $users = User::whereHas('orders', function ($query) {
    $query->where('status', 'paid');
 })->get();
 Có thể hiểu:
 users
 │
 └── kiểm tra orders
       │
       └── có order status = paid?
                │
          ┌─────┴─────┐
         Có          Không
          │             │
      lấy User       bỏ User

 SQL về ý tưởng gần giống:
 SELECT *
 FROM users
 WHERE EXISTS (
    SELECT 1
    FROM orders
    WHERE orders.user_id = users.id
      AND orders.status = 'paid'
 );
 whereHas() không có nghĩa là load orders
 Đây là điểm rất quan trọng.
 $users = User::whereHas('orders', function ($query) {
    $query->where('status', 'paid');
 })->get();
 Nó chỉ nói:
 "Chỉ lấy User nào có order paid."
 Nó không yêu cầu Eloquent lấy danh sách orders về cùng.

 2. with() — Eager Loading
 Ví dụ:
 $users = User::with('orders')->get();
 Nghĩa là:
 Lấy User và load luôn relationship orders.
 Sau đó:
 foreach ($users as $user) {
    foreach ($user->orders as $order) {
        echo $order->status;
    }
 }
 Không bị query orders riêng cho từng user.
 Không dùng with() :
 $users = User::all();
 foreach ($users as $user) {
    echo $user->orders;
 }
 Có thể xảy ra:

 SELECT * FROM users;

 SELECT * FROM orders WHERE user_id = 1;
 SELECT * FROM orders WHERE user_id = 2;
 SELECT * FROM orders WHERE user_id = 3;
 SELECT * FROM orders WHERE user_id = 4;
 ...
 Đây là N+1 Query Problem.

 Dùng with()
 $users = User::with('orders')->get();

 Eloquent về cơ bản làm:
 Query 1:
 SELECT * FROM users;
 
 Query 2:
 SELECT * FROM orders
 WHERE user_id IN (1, 2, 3, 4, ...);

 => Thường chỉ cần 2 query thay vì rất nhiều query.

 Cách dễ bị Race Condition:

 $product = Product::find(1);
 $product->stock = $product->stock - 1;
 $product->save();

 Nó là:
 SELECT stock > PHP tính stock - 1 > UPDATE
 Hai request cùng lúc có thể:
 DB stock = 10

 A đọc 10
 B đọc 10

 A: 10 - 1 = 9
 B: 10 - 1 = 9

 A ghi 9
 B ghi 9

 Kết quả = 9 ❌
 Đúng phải = 8

 Dùng:

 Product::whereKey(1)
     ->decrement('stock');

 Database thực hiện gần như:

 UPDATE products
 SET stock = stock - 1
 WHERE id = 1;

 Hai request:

 stock = 10

 A → DB: stock = stock - 1 → 9
 B → DB: stock = stock - 1 → 8

 Kết quả = 8 ✅

 Đây chính là Atomic Update.

 Nhưng còn một vấn đề: stock âm

 Nếu:

 stock = 0

 mà chạy:

 Product::whereKey(1)
     ->decrement('stock');

 có thể thành:

 stock = -1 ❌

 Vì vậy nghiệp vụ bán hàng nên thêm điều kiện:

 $affected = Product::query()
     ->whereKey($productId)
     ->where('stock', '>=', $quantity)
     ->decrement('stock', $quantity);

 if ($affected === 0) {
     throw new OutOfStockException();
 }

 Ví dụ:

 stock = 1

 A muốn mua 1
 B muốn mua 1

 A:

 UPDATE products
 SET stock = stock - 1
 WHERE id = 1
 AND stock >= 1;
 1 → 0 ✅

 B:

 WHERE stock >= 1

 0 >= 1 ❌

 0 rows affected

 → B nhận OutOfStockException.

 ->increment('stock', 5);

 Cách nhớ:

 increment()
  = DB tự cộng

 decrement()
  = DB tự trừ

 Cả hai

 = tránh kiểu đọc → PHP tính → ghi lại

 Cache cũ hết “thời gian tươi” vẫn tạm trả cho user, trong lúc hệ thống tạo cache mới ở phía sau.
 Nó giúp user không phải đứng chờ việc rebuild cache.

 0 - 300 giây

 → đăng ký refresh sau response

 → phải tính lại
